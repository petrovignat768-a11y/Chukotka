cmake_minimum_required(VERSION 3.20)
project(task_tracker_model VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(task_tracker src/task_tracker.cpp)
target_include_directories(task_tracker PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(task_tracker PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(task_tracker PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(task_tracker_tests tests/task_tracker_test.cpp)
  target_link_libraries(task_tracker_tests PRIVATE task_tracker GTest::gtest_main)
  if(MSVC)
    target_compile_options(task_tracker_tests PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(task_tracker_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  include(GoogleTest)
  gtest_discover_tests(task_tracker_tests)
endif()
